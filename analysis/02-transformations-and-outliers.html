# Create transformed versions
trans_data <- data.frame(
  sale_price = model_df$sale_price,
  log_price = log(model_df$sale_price),
  sqrt_price = sqrt(model_df$sale_price)
)

# Box-Cox transformation
cat("Computing Box-Cox transformation...\n")
Computing Box-Cox transformation...
bc_fit <- boxcox(sale_price ~ 1, data = model_df, lambda = seq(-0.5, 1, 0.1), plotit = FALSE)
lambda_opt <- bc_fit$x[which.max(bc_fit$y)]

if (abs(lambda_opt) < 0.001) {
  trans_data$boxcox_price <- log(model_df$sale_price)
} else {
  trans_data$boxcox_price <- ((model_df$sale_price ^ lambda_opt) - 1) / lambda_opt
}

cat("Optimal Box-Cox lambda:", round(lambda_opt, 4), "\n\n")
Optimal Box-Cox lambda: 0.1 
# Calculate statistics for each transformation
transformations <- list(
  list(name = "Original", data = trans_data$sale_price),
  list(name = "Log", data = trans_data$log_price),
  list(name = "Square Root", data = trans_data$sqrt_price),
  list(name = "Box-Cox", data = trans_data$boxcox_price)
)

results <- data.frame()

for (trans in transformations) {
  mean_val <- mean(trans$data, na.rm = TRUE)
  sd_val <- sd(trans$data, na.rm = TRUE)
  skew_val <- mean((trans$data - mean_val)^3, na.rm = TRUE) / (sd_val^3)

  results <- rbind(results, data.frame(
    Transformation = trans$name,
    Mean = round(mean_val, 2),
    Std_Dev = round(sd_val, 2),
    Skewness = round(skew_val, 4)
  ))
}

3. Outlier Detection

cat("\n===== OUTLIER DETECTION =====\n\n")

===== OUTLIER DETECTION =====
price <- model_df$sale_price

# Method 1: IQR Method
cat("METHOD 1: Interquartile Range (IQR)\n")
METHOD 1: Interquartile Range (IQR)
cat("Rule: Outliers are values outside [Q1 - 1.5*IQR, Q3 + 1.5*IQR]\n\n")
Rule: Outliers are values outside [Q1 - 1.5*IQR, Q3 + 1.5*IQR]
Q1 <- quantile(price, 0.25)
Q3 <- quantile(price, 0.75)
IQR <- Q3 - Q1
lower_iqr <- Q1 - 1.5 * IQR
upper_iqr <- Q3 + 1.5 * IQR

outliers_iqr <- price < lower_iqr | price > upper_iqr
n_outliers_iqr <- sum(outliers_iqr)
pct_iqr <- (n_outliers_iqr / length(price)) * 100

cat("Lower Bound: $", round(lower_iqr, 0), "\n", sep = "")
Lower Bound: $-215008
cat("Upper Bound: $", round(upper_iqr, 0), "\n", sep = "")
Upper Bound: $881004
cat("Outliers Detected:", n_outliers_iqr, "(", round(pct_iqr, 2), "%)\n\n")
Outliers Detected: 9156 ( 4.47 %)
# Method 2: Z-Score Method
cat("METHOD 2: Z-Score (|Z| > 3)\n")
METHOD 2: Z-Score (|Z| > 3)
cat("Rule: Values more than 3 standard deviations from mean\n\n")
Rule: Values more than 3 standard deviations from mean
z_scores <- (price - mean(price)) / sd(price)
outliers_z <- abs(z_scores) > 3
n_outliers_z <- sum(outliers_z)
pct_z <- (n_outliers_z / length(price)) * 100

cat("Outliers Detected:", n_outliers_z, "(", round(pct_z, 2), "%)\n\n")
Outliers Detected: 3447 ( 1.68 %)
# Method 3: Modified Z-Score (MAD - Median Absolute Deviation)
cat("METHOD 3: Modified Z-Score using MAD\n")
METHOD 3: Modified Z-Score using MAD
cat("Rule: |Modified Z| > 3.5, more robust to extreme values\n\n")
Rule: |Modified Z| > 3.5, more robust to extreme values
median_price <- median(price)
mad <- median(abs(price - median_price))
modified_z <- 0.6745 * (price - median_price) / mad
outliers_mad <- abs(modified_z) > 3.5
n_outliers_mad <- sum(outliers_mad)
pct_mad <- (n_outliers_mad / length(price)) * 100

cat("Outliers Detected:", n_outliers_mad, "(", round(pct_mad, 2), "%)\n\n")
Outliers Detected: 7141 ( 3.49 %)
